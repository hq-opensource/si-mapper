    Helper function (module-level, used by both execute_cypher and batch):
    ```python
    def _run_query(driver, query: str) -> dict:
        try:
            records, _, _ = driver.execute_query(query, database_="neo4j")
            rows = [dict(r) for r in records]
            truncated = len(rows) > ROW_LIMIT
            return {"query": query, "result": rows[:ROW_LIMIT], "error": None, "truncated": truncated}
        except Exception as e:
            return {"query": query, "result": None, "error": str(e)}
    ```

    ```python
            os.getenv("NEO4J_PASSWORD", "neo4j_password"),
        )
    ```
