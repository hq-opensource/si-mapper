```python
from typing import Optional
import json
from google.adk.tools import ToolContext

def save_agent_state(tool_context: ToolContext, equipment_id: str, status: str) -> str:
    """Saves the processing status of a specific component to persist state across turns."""
    if "treated" not in tool_context.state:
        tool_context.state["treated"] = {}
    tool_context.state["treated"][equipment_id] = status
    return f":::thought\n[System] Persistent state updated for {equipment_id}: {status}\n:::\n"

def get_agent_state(tool_context: ToolContext, equipment_id: Optional[str] = None) -> str:
    """Retrieves the processing status from the persistent state."""
    treated = tool_context.state.get("treated", {})
    if equipment_id:
        status = treated.get(equipment_id, "unknown")
        return f":::thought\n[System] Status for {equipment_id}: {status}\n:::\n"
    return f":::thought\n[System] Persistent state (treated): {json.dumps(treated)}\n:::\n"
```

**Internal state schema** at `tool_context.state["internal_grid"]`:
```python
{

        # Duct-type (has start/end):
        {"id": "unique-id", "type": "duct", "name": "D-1", "start": [0, 0], "end": [10, 0]},

    ]
```

**Component type categories** (define as module-level constants):
```python
LINE_TYPES = {"duct", "pipe"}  # Components with start/end coords
EQUIPMENT_TYPES = {
    "cooling_coil", "heating_coil", "fan", "filter", "damper",
    "thermal_wheel", "humidifier", "boiler", "heat_pump", "pump",
    "valve_three_way", "valve_two_way", "variable_frequency_drive",
    "room_baseboard", "pipe_chiller"
}
SENSOR_TYPES = {
    "duct_sensor_enthalpy", "duct_sensor_temperature",
    "duct_sensor_differential_pressure", "duct_sensor_humidity",
    "duct_sensor_flow", "duct_sensor_low_limit", "duct_sensor_static_pressure",
    "pipe_sensor_temperature"
}
COORD_TYPES = EQUIPMENT_TYPES | SENSOR_TYPES  # All single-coord types
ALL_TYPES = LINE_TYPES | COORD_TYPES
ROTATION_TYPES = {"fan", "damper"}  # Types that support rotation parameter
```

**Functions to implement:**

1. `initialize_internal_grid(tool_context: ToolContext) -> str`
   - If `"internal_grid"` already exists in `tool_context.state`, return early with message "Internal grid already initialized with N components"
   - Otherwise, set `tool_context.state["internal_grid"] = {"components": []}`
   - Return `":::thought\n[System] Internal grid initialized (empty). Use read_grid MCP tool to load existing state if needed.\n:::"`
   - Note: The agent should call `read_grid` (MCP) first, then use `add_component`/`add_components_batch` to populate initial state from the MCP response. This keeps initialization simple and avoids MCP dependency in the state tools module.

2. `add_component(tool_context: ToolContext, component_type: str, name: str, coord: List[int] = None, start_coord: List[int] = None, end_coord: List[int] = None, rotation: int = 0) -> str`
   - Validate `component_type` is in `ALL_TYPES`; return error string if not
   - Validate coords: if `component_type` in `LINE_TYPES`, require `start_coord` and `end_coord`; if in `COORD_TYPES`, require `coord`
   - Generate a unique ID with `str(uuid.uuid4())[:8]`
   - Build component dict: `{"id": id, "type": component_type, "name": name}` plus coordinate fields based on type
   - For `ROTATION_TYPES`, include `"rotation": rotation`
   - Check for duplicate name in existing components; if found, return error
   - Auto-initialize `internal_grid` if missing (call `initialize_internal_grid` logic inline)
   - Append to `tool_context.state["internal_grid"]["components"]`
   - Return `":::thought\n[System] Added {component_type} '{name}' (id={id}) to internal grid. Total: {N} components.\n:::"`

3. `add_components_batch(tool_context: ToolContext, components: List[Dict]) -> str`
   - Each dict in `components` has keys: `type`, `name`, and either `coord` or `start_coord`/`end_coord`, optionally `rotation`
   - Validate each, skip invalid with error message, add valid ones
   - Auto-initialize if needed
   - Return summary: `":::thought\n[System] Batch add: {added} added, {skipped} skipped. Total: {N} components.\n:::"`

4. `delete_component(tool_context: ToolContext, name: str) -> str`
   - Find component by name in `tool_context.state["internal_grid"]["components"]`
   - Remove it; return error if not found
   - Return `":::thought\n[System] Deleted '{name}' from internal grid. Remaining: {N} components.\n:::"`

5. `delete_components_batch(tool_context: ToolContext, names: List[str]) -> str`
   - Delete all components matching the given names
   - Return summary of deleted vs not-found

6. `read_internal_grid(tool_context: ToolContext, component_type: str = None) -> str`
   - Return JSON string of all components (or filtered by type if specified)
   - Format: `":::thought\n[System] Internal grid ({N} components): {json}\n:::"`
   - If `internal_grid` not initialized, return message saying so

All functions use `from google.adk.tools import ToolContext` and follow the same `:::thought` return format as `state_tools.py`.

Import `uuid`, `json`, `typing.List`, `typing.Dict`, `typing.Optional` at the top. Add logger using `from utils.logging_config import configure_logging`.
